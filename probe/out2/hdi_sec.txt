    <script>
      // Self-heal from a stale index.html. The static site sits behind
      // DigitalOcean's CDN, which caches every URL (query string included) for
      // 24h and cannot be told otherwise. A deploy purges it, but a request in the
      // brief switchover can re-cache the OLD index.html, whose hashed
      // /assets/*.js no longer exist -- on 2026-10-16 that blanked SEC's football
      // embed for its Atlanta visitors until the next deploy. A brand-new query
      // string is always a cache miss, so on a failed /assets/ load reload once
      // with a fresh _r. Also covers a lazy chunk going missing under a tab left
      // open across a deploy (Vite's vite:preloadError).
      //
      // The first retry is immediate (the usual case: a stale CDN copy, deploy long
      // finished). If that also lands in the switchover, back off 2/5/10s (+-1s
      // jitter), then stop. The attempt count rides in the URL (_n), not
      // sessionStorage, which throws inside SEC's sandboxed iframe; an attempt
      // older than 5 min starts over, so a long-lived tab survives the next deploy.
      (function () {
        var DELAYS_MS = [0, 2000, 5000, 10000];
        var RESET_AFTER_MS = 5 * 60 * 1000;
        var handled = false;
        var gaveUp = false;

        function reloadFresh(reason) {
          if (handled) return !gaveUp;
          handled = true;
          var url = new URL(window.location.href);
          var last = Number(url.searchParams.get('_r'));
          var attempt = last && Date.now() - last < RESET_AFTER_MS
            ? Number(url.searchParams.get('_n')) || 0 : 0;
          if (attempt >= DELAYS_MS.length) {
            gaveUp = true;
            console.error('[stale-build] ' + reason + ' -- gave up after ' + attempt + ' reloads.');
            // So a manual refresh starts over (in an iframe it already does).
            try {
              url.searchParams.delete('_r');
              url.searchParams.delete('_n');
              history.replaceState(null, '', url.toString());
            } catch (e) { /* opaque-origin sandbox */ }
            return false;
          }
          // A manual refresh mid-wait lands on that attempt's URL, which the CDN may
          // have cached stale too -- retry at once, or someone refreshing every few
          // seconds never lets the timer fire. Our own retries are 'navigate'.
          var nav = performance.getEntriesByType ? performance.getEntriesByType('navigation')[0] : null;
          var delay = nav && nav.type === 'reload' ? 0 : DELAYS_MS[attempt];
          if (delay) delay += Math.round((Math.random() * 2 - 1) * 1000);
          console.warn('[stale-build] ' + reason + ' -- reload ' + (attempt + 1) + ' in ' + delay + ' ms.');
          setTimeout(function () {
            url.searchParams.set('_r', String(Date.now()));
            url.searchParams.set('_n', String(attempt + 1));
            window.location.replace(url.toString());
          }, delay);
          return true;
        }

        // Resource load errors don't bubble, so listen in the capture phase.
        window.addEventListener('error', function (event) {
          var el = event.target;
          if (!el || el === window) return;
          var src = el.tagName === 'SCRIPT' ? el.src : el.tagName === 'LINK' ? el.href : '';
          if (src && src.indexOf('/assets/') !== -1) reloadFresh('could not load ' + src);
        }, true);

        window.addEventListener('vite:preloadError', function (event) {
          if (reloadFresh('could not load a lazy chunk')) event.preventDefault();
        });
      })();
    </script>

    <script>
      // Conference sites (SEC/ACC) iframe this app's public report/archive embed
      // with sandbox="allow-scripts" but, per their security policy, never
      // "allow-same-origin" (that combo is a known sandbox-escape vector, so their
      // sanitizer strips it whenever both are present -- not something they'll
      // exempt us from). Without allow-same-origin the document is an opaque
      // origin, and merely REFERENCING window.localStorage/sessionStorage throws
      // SecurityError synchronously -- @descope/react-sdk touches
      // window.localStorage at module-import time, before any component renders
      // and before any try/catch in src/ could ever run, so the whole bundle
      // fails to load. This has to run here, as a plain classic script executed
      // before the module bundle below, to install a fallback first.
      //
      // Scoped to the public-embed query-param contract so a locked-down sandbox
      // elsewhere (e.g. someone embedding the full authenticated app) still fails
      // loudly instead of silently degrading with a non-persistent session. Keep
      // this contract check in sync with isPublicEmbedRequest() in src/App.jsx.
      (function () {
        var params = new URLSearchParams(window.location.search);
        var isPublicEmbed = Boolean(
          params.get('conf') && params.get('sport') && params.get('type') && params.get('source')
        );
        if (!isPublicEmbed) return;

        function needsShim(key) {
          try {
            window[key];
            return false;
          } catch (e) {
            return true;
          }
        }

        function makeMemoryStorage() {
          var store = {};
          return {
            getItem: function (k) {
              return Object.prototype.hasOwnProperty.call(store, k) ? store[k] : null;
            },
            setItem: function (k, v) {
              store[k] = String(v);
            },
            removeItem: function (k) {
              delete store[k];
            },
            clear: function () {
              store = {};
            },
            key: function (i) {
              return Object.keys(store)[i] || null;
            },
            get length() {
              return Object.keys(store).length;
            },
          };
        }

        ['localStorage', 'sessionStorage'].forEach(function (key) {
          if (!needsShim(key)) return;
          try {
            Object.defineProperty(window, key, {
              value: makeMemoryStorage(),
              configurable: true,
              writable: true,
            });
            console.warn(
              '[storage-shim] window.' + key + ' unavailable (sandboxed iframe without ' +
              'allow-same-origin) -- using an in-memory fallback for this public embed.'
            );
          } catch (defineError) {
            console.warn('[storage-shim] window.' + key + ' unavailable and could not be shimmed.', defineError);
          }
        });
      })();
    </script>
